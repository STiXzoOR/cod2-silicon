cmake_minimum_required(VERSION 3.20)
project(cod2_platform_tests C CXX OBJC)
if(NOT APPLE)
  message(FATAL_ERROR "These tests exercise native macOS APIs")
endif()
set(CMAKE_OSX_ARCHITECTURES arm64)
set(CMAKE_CXX_STANDARD 17)
set(PORT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_compile_definitions(COD2_X64=1)
link_libraries("-framework IOKit" "-framework CoreFoundation")
add_compile_options(-ffp-contract=off -g)
include_directories("${PORT_ROOT}/src/platform" "${PORT_ROOT}/src" "${PORT_ROOT}/src/headers" "${PORT_ROOT}")
find_package(SDL2 CONFIG REQUIRED)
find_library(OPENGL_FRAMEWORK OpenGL REQUIRED)
find_library(AUDIO_FRAMEWORK AudioToolbox REQUIRED)
find_library(COREAUDIO_FRAMEWORK CoreAudio REQUIRED)
find_library(GC_FRAMEWORK GameController REQUIRED)
find_library(FOUNDATION_FRAMEWORK Foundation REQUIRED)
add_executable(macos_platform_smoke smoke.c
  "${PORT_ROOT}/src/platform/macos_display.c"
  "${PORT_ROOT}/src/platform/macos_rawmouse.m"
  "${PORT_ROOT}/src/platform/macos_audio.c"
  "${PORT_ROOT}/src/platform/macos_system.c")
set_source_files_properties("${PORT_ROOT}/src/platform/macos_rawmouse.m" PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
target_link_libraries(macos_platform_smoke PRIVATE SDL2::SDL2 ${OPENGL_FRAMEWORK} ${AUDIO_FRAMEWORK} ${COREAUDIO_FRAMEWORK} ${GC_FRAMEWORK} ${FOUNDATION_FRAMEWORK})
enable_testing()
add_compile_options(-Wno-typedef-redefinition -Wno-duplicate-decl-specifier
  -Wno-error=implicit-function-declaration -Wno-error=incompatible-pointer-types)
option(COD2_PLATFORM_SANITIZERS "Run portable native adapters under ASan/UBSan" OFF)
if(COD2_PLATFORM_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined)
  add_link_options(-fsanitize=address,undefined)
endif()
add_executable(macos_audio_test audio_test.c "${PORT_ROOT}/src/platform/macos_audio.c")
add_executable(macos_script_notify_test script_notify_test.c)
target_link_options(macos_script_notify_test PRIVATE -Wl,-dead_strip)
add_test(NAME script_notify COMMAND macos_script_notify_test)
add_executable(macos_fx_call_test fx_call_test.c "${PORT_ROOT}/src/PC/universal/com_math.c")
target_link_options(macos_fx_call_test PRIVATE -Wl,-dead_strip)
add_test(NAME fx_call COMMAND macos_fx_call_test)
add_executable(macos_slidemove_test slidemove_test.c)
target_link_options(macos_slidemove_test PRIVATE -Wl,-dead_strip)
add_test(NAME slidemove COMMAND macos_slidemove_test)
add_executable(macos_random_test random_test.c)
target_link_options(macos_random_test PRIVATE -Wl,-dead_strip)
add_test(NAME random COMMAND macos_random_test)
add_executable(macos_cgame_layout_test cgame_layout.c)
target_link_options(macos_cgame_layout_test PRIVATE -Wl,-dead_strip)
add_test(NAME cgame_layout COMMAND macos_cgame_layout_test)
add_executable(macos_snd_channel_test snd_channel.c)
target_link_options(macos_snd_channel_test PRIVATE -Wl,-dead_strip)
add_test(NAME snd_channel COMMAND macos_snd_channel_test)
add_executable(macos_jpeg_test jpeg_test.c "${PORT_ROOT}/src/platform/macos_jpeg.c")
target_link_libraries(macos_jpeg_test PRIVATE "-framework ImageIO" "-framework CoreGraphics" "-framework CoreFoundation")
add_test(NAME jpeg COMMAND macos_jpeg_test)
add_executable(macos_hardware_test hardware_test.c "${PORT_ROOT}/src/platform/macos_system.c" "${PORT_ROOT}/src/platform/macos_display.c")
target_link_libraries(macos_hardware_test PRIVATE SDL2::SDL2 ${OPENGL_FRAMEWORK})
add_test(NAME hardware COMMAND macos_hardware_test)
add_executable(macos_gamma_test gamma.c "${PORT_ROOT}/src/platform/macos_system.c")
target_link_libraries(macos_gamma_test PRIVATE SDL2::SDL2 ${OPENGL_FRAMEWORK})
add_test(NAME presentation_gamma COMMAND macos_gamma_test)
set_tests_properties(presentation_gamma PROPERTIES RUN_SERIAL TRUE)
add_executable(macos_voice_test voice_test.c "${PORT_ROOT}/src/platform/macos_audio.c" "${PORT_ROOT}/src/platform/macos_voice.c")
foreach(target macos_audio_test macos_voice_test)
  target_link_libraries(${target} PRIVATE ${AUDIO_FRAMEWORK} ${COREAUDIO_FRAMEWORK})
endforeach()
add_executable(macos_cpp_abi_test cpp_abi.cpp "${PORT_ROOT}/src/platform/macos_cpp_abi.cpp" "${PORT_ROOT}/src/PC/stringed/stringed_interface.c")
target_compile_definitions(macos_cpp_abi_test PRIVATE COD2_CPP_TEST_STRINGED=1)
add_executable(macos_threads_ring_test threads_ring.c "${PORT_ROOT}/src/platform/macos_threads.c" "${PORT_ROOT}/src/platform/macos_ring.c"
  "${PORT_ROOT}/src/Mac/Tools/MacWin32.c" "${PORT_ROOT}/src/platform/macos_system.c")
target_link_options(macos_threads_ring_test PRIVATE -Wl,-dead_strip)
add_executable(macos_input_test input.c "${PORT_ROOT}/src/unix/linux_input.c" "${PORT_ROOT}/src/platform/macos_rawmouse.m")
set_source_files_properties("${PORT_ROOT}/src/unix/linux_input.c" PROPERTIES COMPILE_DEFINITIONS "SDL_PollEvent=MacTest_PollEvent;SDL_PeepEvents=MacTest_PeepEvents;SDL_PumpEvents=MacTest_PumpEvents;SDL_GetTicks=MacTest_GetTicks;SDL_GetWindowFlags=MacTest_GetWindowFlags")
target_link_libraries(macos_input_test PRIVATE SDL2::SDL2 ${GC_FRAMEWORK} ${FOUNDATION_FRAMEWORK})
add_executable(macos_system_network_test macos_system_network.c
  "${PORT_ROOT}/src/Mac/Main/mac_main.c" "${PORT_ROOT}/src/Mac/Main/mac_common.c"
  "${PORT_ROOT}/src/Mac/Tools/MacWin32.c" "${PORT_ROOT}/src/PC/win32/win_shared.c"
  "${PORT_ROOT}/src/unix/linux_net.c" "${PORT_ROOT}/src/platform/macos_system.c")
target_compile_definitions(macos_system_network_test PRIVATE DEDICATED=1)
target_link_options(macos_system_network_test PRIVATE -Wl,-dead_strip)
if(NOT COD2_PLATFORM_SANITIZERS)
  add_executable(macos_diagnostics_test macos_diagnostics.c "${PORT_ROOT}/src/platform/macos_system.c")
  target_compile_definitions(macos_diagnostics_test PRIVATE COD2_GUARDHEAP=1)
  target_link_options(macos_diagnostics_test PRIVATE -Wl,-dead_strip)
  add_test(NAME diagnostics COMMAND macos_diagnostics_test)
  add_executable(macos_crash_test macos_crash.c "${PORT_ROOT}/src/PC/qcommon/crash_handler.c")
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME crash COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/check_crash.py" "$<TARGET_FILE:macos_crash_test>")
endif()
add_executable(macos_dedicated_fx_test macos_dedicated_fx.c
  "${PORT_ROOT}/src/PC/EffectsCore/GenericParser2.c" "${PORT_ROOT}/src/PC/EffectsCore/FxTemplate.c"
  "${PORT_ROOT}/src/PC/EffectsCore/FxScheduler.c" "${PORT_ROOT}/src/PC/EffectsCore/FxScheduler_load_obj.c"
  "${PORT_ROOT}/src/PC/EffectsCore/FxChannel.c" "${PORT_ROOT}/src/PC/EffectsCore/FxCurve.c"
  "${PORT_ROOT}/src/PC/EffectsCore/FxCurve_load_obj.c")
target_compile_definitions(macos_dedicated_fx_test PRIVATE DEDICATED=1)
target_link_options(macos_dedicated_fx_test PRIVATE -Wl,-dead_strip)
add_test(NAME dedicated_fx COMMAND macos_dedicated_fx_test)
get_target_property(MACOS_FX_TEST_SOURCES macos_dedicated_fx_test SOURCES)
add_executable(macos_client_fx_test ${MACOS_FX_TEST_SOURCES})
target_link_options(macos_client_fx_test PRIVATE -Wl,-dead_strip)
add_test(NAME client_fx COMMAND macos_client_fx_test)
add_test(NAME smoke_windowed COMMAND macos_platform_smoke)
add_test(NAME smoke_borderless COMMAND macos_platform_smoke --borderless)
add_test(NAME smoke_fullscreen COMMAND macos_platform_smoke --fullscreen)
add_test(NAME input COMMAND macos_input_test)
add_test(NAME threads_ring COMMAND macos_threads_ring_test)
add_test(NAME cpp_abi COMMAND macos_cpp_abi_test)
add_test(NAME audio COMMAND macos_audio_test)
add_test(NAME voice COMMAND macos_voice_test)
add_test(NAME system_network COMMAND macos_system_network_test)
add_test(NAME system_closed_stdin COMMAND macos_system_network_test --closed-stdin)
add_test(NAME system_no_home COMMAND "${CMAKE_COMMAND}" -E env --unset=HOME "$<TARGET_FILE:macos_system_network_test>")
# Window focus and the default audio device are shared resources on the test Mac.
set_tests_properties(smoke_windowed smoke_borderless smoke_fullscreen input audio voice PROPERTIES RUN_SERIAL TRUE)
